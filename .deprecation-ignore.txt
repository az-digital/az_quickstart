# Deprecation patterns to ignore when running Quickstart's PHPUnit tests.
#
# Drupal core's DeprecationBridge test extension reads this file from the
# SYMFONY_DEPRECATIONS_HELPER environment variable, which is set in
# .github/workflows/ci.yml and .ddev/config.quickstart.yaml:
#   SYMFONY_DEPRECATIONS_HELPER="ignoreFile=/path/to/az_quickstart/.deprecation-ignore.txt"
# Each non-comment line is a PCRE pattern matched against deprecation messages.
# See https://www.drupal.org/node/3285162
#
# phpunit.xml.dist sets ignoreSuppressionOfDeprecations="true" so that
# deprecations Drupal reports with @trigger_error() also fail tests. That
# includes deprecations raised by the test site during functional tests.

# Only fail on deprecations in Quickstart's own code: ignore any deprecation
# whose message doesn't mention a Quickstart extension (az_*). Deprecated core
# APIs that Quickstart calls are still caught by PHPStan deprecation rules.
%^(?!.*(?:^|[^a-zA-Z0-9])az_[a-z0-9]).*$%s

# az_migration is intentionally deprecated.
# See https://github.com/az-digital/az_quickstart/issues/5332
%^The module 'az_migration' is deprecated\.%
